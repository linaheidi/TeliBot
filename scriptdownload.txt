     * The page includes the Monetag SDK script tag as requested.

    function openAdFor(fileId){
      const file = appData.files.find(f=>f.id===fileId);
      if(!file) return alert('File not found');

      // If SDK provides a show function named in data-sdk attribute we try that.
      // The script tag had data-sdk="show_9837009" so try window.show_9837009
      const sdkFn = window.show_9837009 || window.show_9837009 === 0 ? window.show_9837009 : null;

      // create a simple overlay/modal indicating ad is displayed
      showAdOverlay();

      if(typeof sdkFn === 'function'){
        // try to call SDK; some SDKs accept a callback param or return a Promise.
        // We'll attempt several patterns; since we don't know exact API, handle gracefully.
        try{
          // pattern 1: sdkFn(callback)
          let called = false;
          const maybeResult = sdkFn(function(sdkEvent){
            called = true;
            hideAdOverlay();
            // if sdkEvent indicates finished
            // We'll assume any callback = ad viewed; increment by 1
            incrementAdCount(fileId, 1, 'sdk-callback');
          });
          // pattern 2: sdkFn() might open and call global event later
          if(maybeResult && maybeResult.then){
            // promise style
            maybeResult.then(() => {
              hideAdOverlay();
              incrementAdCount(fileId,1,'sdk-promise');
            }).catch(err => {
              hideAdOverlay();
              console.warn('SDK promise rejected',err);
              // fallback to simulated
              simulateAdView(fileId);
            });
          } else {
            // If callback not called within X seconds, fallback to simulate
            setTimeout(()=> {
              if(!called){
                // fallback: maybe SDK will call later; keep overlay for some more seconds
                // But don't hang forever — simulate after 8s
                setTimeout(()=> {
                  if(document.body.contains(adOverlay)) return simulateAdView(fileId);
                }, 8000);
              }
            }, 1200);
          }
        }catch(err){
          console.warn('SDK call error',err);
          hideAdOverlay();
          simulateAdView(fileId);
        }
      } else {
        // SDK not available: simulate an ad (demo)
        simulateAdView(fileId);
      }
    }
